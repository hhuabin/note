# `jwt` 基本使用

1.   包引入

     ```xml
     <properties>
         <java.version>17</java.version>
         <!-- Spring Boot 2.7.18 版本 -->
         <jjwt.version>0.11.5</jjwt.version>
     </properties>
     
     <!-- JWT API -->
     <dependency>
         <groupId>io.jsonwebtoken</groupId>
         <artifactId>jjwt-api</artifactId>
         <version>${jjwt.version}</version>
     </dependency>
     
     <!-- JWT 实现 -->
     <dependency>
         <groupId>io.jsonwebtoken</groupId>
         <artifactId>jjwt-impl</artifactId>
         <version>${jjwt.version}</version>
         <scope>runtime</scope>
     </dependency>
     
     <!-- JWT JSON 序列化 -->
     <dependency>
         <groupId>io.jsonwebtoken</groupId>
         <artifactId>jjwt-jackson</artifactId>
         <version>${jjwt.version}</version>
         <scope>runtime</scope>
     </dependency>
     ```

2.   `jwt` 全局配置（`application.yml`）

     ```yaml
     # JWT 配置
     jwt:
       # Base64 编码的随机密钥
       secret: ${JWT_SECRET}
       # Token 有效期：2小时
       expiration: 7200000
       # 请求头名称
       header: Authorization
       # Token 前缀
       prefix: Bearer
     ```

3.   配置 `JWT` 工具类

     ```java
     @Component
     public class JwtUtil {
     
         /**
          * 签名用的密钥，来自配置文件 jwt.secret，至少要 32 个字符
          */
         private final SecretKey key;
     
         /**
          * 票的有效期，单位秒，来自 jwt.expire-seconds。
          * 默认 64800 秒 = 18 小时，配置缺失时用它兜底
          */
         private final long expireSeconds;
     
         public JwtUtil(@Value("${jwt.secret}") String secret,
                        @Value("${jwt.expire-seconds:64800}") long expireSeconds) {
             // HS256 要求密钥不少于 32 字节，太短这里会直接启动失败
             this.key = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
             this.expireSeconds = expireSeconds;
         }
     
         /**
          * 1. 根据 userId 生成 token。
          * 登录成功时调用一次。
          *
          * @param userId 用户 id
          * @return 签名后的 token 字符串
          */
         public String generateToken(Long userId) {
             Date now = new Date();
     
             return Jwts.builder()
                     // 票上写"我是谁"，只放 userId，不放密码等敏感信息
                     .setSubject(String.valueOf(userId))
                     // 发票时间
                     .setIssuedAt(now)
                     // 过期时间 = 现在 + 有效期
                     .setExpiration(new Date(now.getTime() + expireSeconds * 1000))
                     // 用密钥盖章，别人改一个字符章就对不上
                     .signWith(key, SignatureAlgorithm.HS256)
                     .compact();
         }
     
         /**
          * 2. 根据 token 获取用户 id。
          * 签名不对或已过期都会抛异常，所以调用方要么 try-catch，要么先用方法 3 判断。
          *
          * @param token 前端放在请求头里的 token（不含 Bearer 前缀）
          * @return 票上的用户 id
          * @throws JwtException 签名不合法、格式错误或已过期
          */
         public Long getUserId(String token) {
             return Long.valueOf(parseClaims(token).getSubject());
         }
     
         /**
          * 3. 校验 token 是否已过期。
          * 注意：jjwt 解析到过期 token 时会直接抛 ExpiredJwtException，
          * 拿不到票面内容，所以必须在这里接住它并返回 true。
          *
          * @param token 前端放在请求头里的 token（不含 Bearer 前缀）
          * @return true 表示已过期或根本解析不了（不能用），false 表示还在有效期内
          */
         public boolean isExpired(String token) {
             try {
                 Date expiration = parseClaims(token).getExpiration();
                 return expiration.before(new Date());
             } catch (ExpiredJwtException e) {
                 // 明确告诉我们是过期，这就是要判断的情况
                 return true;
             } catch (JwtException | IllegalArgumentException e) {
                 // 签名被改过、格式乱、token 为空等，同样视为不能用
                 return true;
             }
         }
     
         /**
          * 验章并取出票面内容。方法 2 和方法 3 共用，避免重复写解析代码。
          */
         private Claims parseClaims(String token) {
             return Jwts.parserBuilder()
                     // 用同一把密钥验章
                     .setSigningKey(key)
                     .build()
                     .parseClaimsJws(token)
                     .getBody();
         }
     }
     
     ```
