#  小程序生命周期（`app.js`）

[微信小程序官方文档](https://developers.weixin.qq.com/miniprogram/dev/reference/api/App.html "微信小程序官方文档")

| 生命周期               | 触发时机                               | 参数      |
| ---------------------- | -------------------------------------- | --------- |
| `onLaunch`             | 小程序初始化完成时触发，**只触发一次** | `options` |
| `onShow`               | 小程序启动、从后台进入前台             | `options` |
| `onHide`               | 小程序从前台进入后台                   | 无        |
| `onError`              | 脚本错误、API 调用报错                 | `msg`     |
| `onUnhandledRejection` | 未处理的 Promise rejection             | `res`     |
| `onPageNotFound`       | 打开不存在的页面                       | `res`     |

```typescript
// app.js
App({
    // 全局数据
    globalData: {
        userInfo: null,
    },
    onLaunch(options) {
        /**
         * @description 在启动时执行
         * @param query 启动时的参数
         * @param scene 启动时的场景
         * @param path 启动时的路径
         * @param referrerInfo 启动时的来源信息
         */
        const { query, scene, path, referrerInfo } = options
    },

    // 小程序启动，或从后台进入前台时触发
    onShow(options) {
        console.log('onShow', options)
    },

    // 小程序从前台进入后台时触发
    onHide() {
        console.log('onHide')
    },

    // 小程序发生脚本错误时触发
    onError(msg) {
        console.log('onError', msg)
    },

    // 小程序发生未处理的 Promise rejection 时触发
    onUnhandledRejection(res) {
        console.log('onUnhandledRejection', res)
    },

    // 小程序打开了不存在的页面时触发
    onPageNotFound(res) {
        console.log('onPageNotFound', res)
    },

    /**
     * @description 系统主题发生变化时触发
     * @param res 主题变化信息
     * @param res.theme 当前主题，light / dark
     */
    onThemeChange(res) {
        console.log('onThemeChange', res.theme)
    },
})

```

### 新的推荐写法

```typescript
App()

wx.onError(function (msg) {
    // 发生脚本错误时执行
    console.log(msg)

func(options)
```



# 页面生命周期

[页面生命周期文档](https://developers.weixin.qq.com/miniprogram/dev/reference/api/Page.html "页面生命周期")

| 方法                | 含义                                                         | 类型         |
| ------------------- | ------------------------------------------------------------ | ------------ |
| `onLoad`            | 页面加载时触发，一个页面只触发一次                           | 页面生命周期 |
| `onShow`            | 页面显示/进入前台时触发                                      | 页面生命周期 |
| `onReady`           | 页面初次渲染完成时触发，一个页面只触发一次                   | 页面生命周期 |
| `onHide`            | 页面隐藏时触发                                               | 页面生命周期 |
| `onUnload`          | 页面卸载时触发                                               | 页面生命周期 |
| `onPullDownRefresh` | 用户下拉刷新时触发                                           | 页面事件     |
| `onReachBottom`     | 页面滚动到页面底部时触发                                     | 页面事件     |
| `onPageScroll`      | 页面滚动时触发                                               | 页面事件     |
| `onResize`          | 页面尺寸发生变化时触发                                       | 页面事件     |
| `onShareAppMessage` | 用户点击页面内转发按钮或右上角转发时触发，用于自定义转发内容 | 页面事件     |
| `onShareTimeline`   | 用户将页面分享到朋友圈时触发，用于自定义分享到朋友圈的内容   | 页面事件     |
| `onAddToFavorites`  | 用户将页面添加到收藏时触发，用于自定义收藏内容               | 页面事件     |
| `onTabItemTap`      | 用户点击 TabBar 时触发                                       | 页面事件     |

```typescript
Page({
    data: {
        text: "This is page data."
    },
    onLoad: function (options) {
        // Do some initialize when page load.
    },
    onShow: function () {
        // Do something when page show.
    },
    onReady: function () {
        // Do something when page ready.
    },
    onHide: function () {
        // Do something when page hide.
    },
    onUnload: function () {
        // Do something when page close.
    },
    onPullDownRefresh: function () {
        // Do something when pull down.
    },
    onReachBottom: function () {
        // Do something when page reach bottom.
    },
    onShareAppMessage: function () {
        // return custom share data when user share.
    },
    onPageScroll: function () {
        // Do something when page scroll
    },
    onResize: function () {
        // Do something when page resize
    },
})
```



# 组件生命周期

[组件生命周期文档](https://developers.weixin.qq.com/miniprogram/dev/framework/custom-component/lifetimes.html "组件生命周期")

```typescript
Component({
    // 组件可以监听所在页面的生命周期
    pageLifetimes: {
        show() {
            console.log('组件所在页面显示')
        },

        hide() {
            console.log('组件所在页面隐藏')
        },

        resize(size) {
            console.log('组件所在页面尺寸变化', size)
        },

        routeDone() {
            console.log('组件所在页面路由动画完成')
        },
    },
    // 组件生命周期
    lifetimes: {
        created() {
            console.log('组件创建')
        },

        attached() {
            console.log('组件进入页面节点树')
        },

        ready() {
            console.log('组件渲染完成')
        },

        moved() {
            console.log('组件位置发生变化')
        },

        detached() {
            console.log('组件从页面移除')
        },

        error(err) {
            console.log('组件发生错误', err)
        },
    },
})
```
