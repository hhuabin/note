# 访问 `/` 得到前端项目

在 `Linux` 中，前端项目一般放在，适合较多，任君选择

| 目录                     | 常见用途                               |
| ------------------------ | -------------------------------------- |
| `/var/www/`              | ⭐ 很适合放网站                         |
| `/usr/share/nginx/html/` | Nginx 默认网站目录                     |
| `/opt/`                  | 企业应用、第三方软件等                 |
| `/data/`                 | 企业自己规划的数据/应用目录            |
| `/home/xxx/`             | 用户自己的文件，一般不作为正式网站目录 |

```shell
mkdir /var/www
```

`SPA` 最重要坑：`index.html` 绝对不能长缓存，直接取消缓存

1. 配置 `try_files`

    ```nginx
    server {
        listen       80;
        server_name  localhost;
    
        location / {
            root /var/www/my-project;             # 项目在linux中的位置(/var/www/my-project/index.html)
            try_files $uri $uri/ /index.html;     # 尝试访问请求的 URI，如果失败，则尝试访问目录形式的 URI，最后如果仍然无法匹配，则重写为 index.html
            index index.html;
        }
    }

2. 对于多个项目

    不同的项目可以运行在不同的端口号中，如此，一个域名下即可运行多个项目

    ```nginx
    server {
        listen       443;
        server_name  hhuabin;
    
        location / {
            root /var/www/my-project;       # 项目在linux中的位置(/var/www/my-project/index.html)
            try_files $uri $uri/ /index.html;
    		index index.html;
        }
    }
    ```



# 访问 `/projectname` 得到前端项目

1. 直接重定向至项目的端口号（访问`/`）

   ```nginx
   server {
       listen       80;
       server_name  localhost;
   
       location / {
           return 301 http://192.168.198.128:443/;
       }
   }
   ```

2. 对于**多个项目部署在同一个端口号**上的情况（例如默认的80端口）

   `nginx` 的配置

   ```nginx
   server {
       listen       80;
       server_name  localhost;
   
       location /projectname1/ {
           root /var/www;                # 项目在linux中的位置(/var/www/projectname1/index.html)
           try_files $uri $uri/ /index.html;
   		index index.html;
       }
   
       location /projectname2/ {
           root /var/www;                # 项目在linux中的位置(/var/www/projectname2/index.html)
           try_files $uri $uri/ /index.html;
   		index index.html;
       }
   }
   ```
   
   此时前端同样需要做出一定的调整（此处示例为 `vite`）
   
   1. `vite.config.ts`
   
      ```ts
      export default defineConfig({
          base: '/projectname1/',            // 此处的 projectname1 需要和上面保持一致
          build: {
              outDir: 'dist/projectname1',   // 此处可以不配置，但是为了方便本地运行，建议加上，但是部署到后端应该使用 projectname1 文件夹里面的即；配置了 CI/CD 不能写该句
          }
      })
      ```
   
   2. `React18` 路由 `router/index.tsx`

      ```tsx
      export const router = createBrowserRouter(
          routes,
          {
              basename: import.meta.env.BASE_URL.replace(/\/$/, '') || '/',   // import.meta.env.BASE_URL = /react18/
          },
      )
      ```
   
   3. `Vue3` 路由 `router.js`
   
      ```ts
      const router = createRouter({
          history: createWebHistory(import.meta.env.BASE_URL),  // import.meta.env.BASE_URL = /projectname1/
      })
      ```
   



# 提供一个访问 `/react18/` 得到前端项目实际 `nginx` 配置

```nginx
http {
    include       mime.types;
    default_type  application/octet-stream;

    #log_format  main  '$remote_addr - $remote_user [$time_local] "$request" '
    #                  '$status $body_bytes_sent "$http_referer" '
    #                  '"$http_user_agent" "$http_x_forwarded_for"';

    #access_log  logs/access.log  main;

    sendfile        on;
    #tcp_nopush     on;

    #keepalive_timeout  0;
    keepalive_timeout  65;

    #gzip  on;

    server {
        listen       80;
        server_name  localhost;

        #access_log  logs/host.access.log  main;

        location / {
            root   html;
            index  index.html index.htm;
        }

        #error_page  404              /404.html;

        # redirect server error pages to the static page /50x.html
        #
        error_page   500 502 503 504  /50x.html;
        location = /50x.html {
            root   html;
        }

        
        # 前端项目配置
        # /react18 -> /react18/；/react18 是具体路径，nginx 不会默认加 /，所以需要配置重定向到/
        location = /react18 {
            return 301 /react18/;
        }
        
        # index.html 禁止缓存
        location = /react18/index.html {
            root /var/www;

            add_header Cache-Control "no-cache, no-store, must-revalidate";
            add_header Pragma "no-cache";
            add_header Expires "0";
        }
        
        # 静态资源：缓存 7 天
        location /react18/assets/ {
            root /var/www;

            expires 7d;
            add_header Cache-Control "public";
        }

        # React History 路由
        location /react18/ {
            root /var/www;

            # History 路由
            try_files $uri $uri/ /react18/index.html;
        }


    }
}
```

不能这样子`location /react18`配置，它甚至会匹配上 `/react18abc`

```nginx
location /react18 {
    root /var/www;

    # History 路由
    try_files $uri $uri/ /react18/index.html;
}
```

# 每次更新配置都要重新加载配置


sudo /usr/local/nginx/sbin/nginx -s reload
```
